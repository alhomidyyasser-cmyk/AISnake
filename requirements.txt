# Runtime dependencies for AI Snake.

# `mp.solutions` API used by cvzone's HandDetector. MediaPipe and the

# to keep a fresh install reproducible.

# Pinning both to the same 4.11.x line avoids mismatched cv2 builds.
opencv-contrib-python>=4.8,<4.12
cvzone>=1.6,<2
mediapipe==0.10.21
numpy>=1.24,<2
